import os, tempfile, time
from codecarbon import OfflineEmissionsTracker
import pandas as pd
for mode in ("process", "machine"):
    out = tempfile.mkdtemp()
    t = OfflineEmissionsTracker(country_iso_code="ESP", tracking_mode=mode, measure_power_secs=15, output_dir=out, log_level="warning")
    cpu0, wall0 = time.process_time(), time.perf_counter()
    with t:
        end = time.process_time() + 3
        while time.process_time() < end:
            pass
    wall, cpu = time.perf_counter() - wall0, time.process_time() - cpu0
    r = pd.read_csv(os.path.join(out, "emissions.csv")).iloc[-1]
    h = next(h for h in t._hardware if type(h).__name__ == "CPU")
    print(f"{mode}: method={h._mode} tdp={h._tdp} generic={h._is_generic_tdp} cpu_model={r['cpu_model']!r} cpu_count={r['cpu_count']} ram_gb={r['ram_total_size']:.1f} wall={wall:.2f}s cpu={cpu:.2f}s cpu_W={r['cpu_energy']*3.6e6/r['duration']:.2f} ram_W={r['ram_energy']*3.6e6/r['duration']:.2f} total_W={r['energy_consumed']*3.6e6/r['duration']:.2f}")
